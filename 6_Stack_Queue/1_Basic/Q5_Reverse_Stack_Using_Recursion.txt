Question Number: 5
Problem Title: Reverse a stack using recursion
Problem Description:
Given a stack, reverse it using recursion only.

Approach / Explanation:
1) Pop the top element recursively until stack is empty.
2) Insert each element at the bottom using another recursive helper.
3) This reverses the order without using extra stack structures.
4) Uses call stack as extra space.
5) Demonstrates recursion with stack operations.
6) Time complexity is O(n^2) due to repeated insert at bottom.
7) Output stack content after reversal.

Brute Force Note:
Using an auxiliary stack is O(n) and simpler.

C++ Code:
#include <bits/stdc++.h>
using namespace std;

void insertBottom(stack<long long>& st, long long x) {
    if (st.empty()) { st.push(x); return; }
    long long top = st.top(); st.pop();
    insertBottom(st, x);
    st.push(top);
}

void reverseStack(stack<long long>& st) {
    if (st.empty()) return;
    long long x = st.top(); st.pop();
    reverseStack(st);
    insertBottom(st, x);
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n; if (!(cin >> n) || n < 0) return 0;
    stack<long long> st;
    for (int i = 0; i < n; ++i) { long long x; cin >> x; st.push(x); }

    reverseStack(st);

    vector<long long> out;
    while (!st.empty()) { out.push_back(st.top()); st.pop(); }
    for (int i = 0; i < (int)out.size(); ++i) {
        if (i) cout << " ";
        cout << out[i];
    }
    return 0;
}

Time Complexity: O(n^2)
Space Complexity: O(n) recursion
