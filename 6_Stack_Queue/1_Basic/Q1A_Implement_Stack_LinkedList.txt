Question Number: 1A
Problem Title: Implement a stack using linked list
Problem Description:
Implement a stack with push, pop, top, and isEmpty operations using a linked list.

Approach / Explanation:
1) Use a linked list with head as the stack top.
2) Push inserts a node at head.
3) Pop removes the head node.
4) Top reads the head value.
5) isEmpty checks if head is null.
6) All operations are O(1).
7) Output results for sample operations.

Brute Force Note:
Array-based stack is simpler but has fixed capacity.

C++ Code:
#include <bits/stdc++.h>
using namespace std;

struct Node {
    long long val;
    Node* next;
    Node(long long v) : val(v), next(nullptr) {}
};

struct StackLL {
    Node* head = nullptr;
    bool isEmpty() const { return head == nullptr; }
    void push(long long x) { Node* n = new Node(x); n->next = head; head = n; }
    bool pop() {
        if (isEmpty()) return false;
        Node* t = head; head = head->next; delete t; return true;
    }
    bool peek(long long &out) const {
        if (isEmpty()) return false;
        out = head->val; return true;
    }
};

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    StackLL st;
    int q; if (!(cin >> q)) return 0;
    while (q--) {
        string op; cin >> op;
        if (op == "push") {
            long long x; cin >> x;
            st.push(x); cout << "ok\n";
        } else if (op == "pop") {
            cout << (st.pop() ? "ok" : "underflow") << "\n";
        } else if (op == "top") {
            long long v;
            if (st.peek(v)) cout << v << "\n"; else cout << "empty\n";
        } else if (op == "empty") {
            cout << (st.isEmpty() ? 1 : 0) << "\n";
        }
    }

    return 0;
}

Time Complexity: O(1) per operation
Space Complexity: O(n)
