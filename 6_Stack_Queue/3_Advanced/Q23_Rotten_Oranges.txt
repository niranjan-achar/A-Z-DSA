Question Number: 23
Problem Title: Rotten Oranges (Multi-source BFS using Queue)
Problem Description:
Given a grid with fresh and rotten oranges, find minimum time to rot all.

Approach / Explanation:
1) Push all rotten oranges into a queue with time 0.
2) Perform BFS in 4 directions.
3) When a fresh orange is adjacent, rot it and push with time+1.
4) Track the maximum time encountered.
5) If any fresh orange remains, return -1.
6) Multi-source BFS ensures minimal time.
7) Output the time or -1.

Brute Force Note:
Simulating minute by minute without BFS is inefficient.

C++ Code:
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int r, c; if (!(cin >> r >> c) || r <= 0 || c <= 0) return 0;
    vector<vector<int>> grid(r, vector<int>(c));
    queue<pair<int,int>> q;
    int fresh = 0;
    for (int i = 0; i < r; ++i) {
        for (int j = 0; j < c; ++j) {
            cin >> grid[i][j];
            if (grid[i][j] == 2) q.push({i, j});
            else if (grid[i][j] == 1) ++fresh;
        }
    }

    int minutes = 0;
    int dirs[4][2] = {{1,0},{-1,0},{0,1},{0,-1}};
    while (!q.empty() && fresh > 0) {
        int sz = (int)q.size();
        while (sz--) {
            auto [x, y] = q.front(); q.pop();
            for (auto &d : dirs) {
                int nx = x + d[0], ny = y + d[1];
                if (nx>=0 && nx<r && ny>=0 && ny<c && grid[nx][ny]==1) {
                    grid[nx][ny] = 2;
                    --fresh;
                    q.push({nx, ny});
                }
            }
        }
        ++minutes;
    }

    cout << (fresh == 0 ? minutes : -1);
    return 0;
}

Time Complexity: O(r*c)
Space Complexity: O(r*c)
