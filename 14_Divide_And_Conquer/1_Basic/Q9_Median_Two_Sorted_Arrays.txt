Problem 9: Median of Two Sorted Arrays

Description:
Find the median of two sorted arrays of different sizes. Median is the middle value(s).

Approach:
Use binary search on the smaller array to partition both arrays such that left partition size = right partition size. Compare boundary elements.

Brute Force Note:
Merge arrays and sort: O((m+n) log(m+n)) time.

C++ Code:
```cpp
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

double findMedianSortedArrays(vector<int>& arr1, vector<int>& arr2) {
    if (arr1.size() > arr2.size())
        return findMedianSortedArrays(arr2, arr1);
    
    int m = arr1.size(), n = arr2.size();
    int low = 0, high = m;
    
    while (low <= high) {
        int cut1 = low + (high - low) / 2;
        int cut2 = (m + n + 1) / 2 - cut1;
        
        int left1 = (cut1 == 0) ? INT_MIN : arr1[cut1 - 1];
        int left2 = (cut2 == 0) ? INT_MIN : arr2[cut2 - 1];
        int right1 = (cut1 == m) ? INT_MAX : arr1[cut1];
        int right2 = (cut2 == n) ? INT_MAX : arr2[cut2];
        
        if (left1 <= right2 && left2 <= right1) {
            if ((m + n) % 2 == 0)
                return (max(left1, left2) + min(right1, right2)) / 2.0;
            else
                return max(left1, left2);
        } else if (left1 > right2) {
            high = cut1 - 1;
        } else {
            low = cut1 + 1;
        }
    }
    
    return -1;
}

int main() {
    vector<int> arr1 = {1, 3};
    vector<int> arr2 = {2};
    
    cout << "Median: " << findMedianSortedArrays(arr1, arr2) << endl;
    
    return 0;
}
```

Time Complexity: O(log(min(m, n))) - Binary search on smaller array
Space Complexity: O(1) - Constant space
