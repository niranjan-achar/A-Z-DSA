Problem 3: Merge Sort

Description:
Sort an array using the Merge Sort algorithm which divides the array recursively and merges sorted subarrays.

Approach:
Divide the array into two halves recursively until single elements remain. Merge the sorted halves back together in sorted order.

Brute Force Note:
Bubble sort or selection sort: O(n²) time.

C++ Code:
```cpp
#include <iostream>
#include <vector>
using namespace std;

void merge(vector<int>& arr, int left, int mid, int right) {
    int n1 = mid - left + 1;
    int n2 = right - mid;
    
    vector<int> leftArr(n1), rightArr(n2);
    
    for (int i = 0; i < n1; i++)
        leftArr[i] = arr[left + i];
    for (int i = 0; i < n2; i++)
        rightArr[i] = arr[mid + 1 + i];
    
    int i = 0, j = 0, k = left;
    
    while (i < n1 && j < n2) {
        if (leftArr[i] <= rightArr[j]) {
            arr[k++] = leftArr[i++];
        } else {W
            arr[k++] = rightArr[j++];
        }
    }
    
    while (i < n1)
        arr[k++] = leftArr[i++];
    while (j < n2)
        arr[k++] = rightArr[j++];
}

void mergeSort(vector<int>& arr, int left, int right) {
    if (left < right) {
        int mid = left + (right - left) / 2;
        mergeSort(arr, left, mid);
        mergeSort(arr, mid + 1, right);
        merge(arr, left, mid, right);
    }
}

int main() {
    vector<int> arr = {64, 34, 25, 12, 22, 11, 90};
    mergeSort(arr, 0, arr.size() - 1);
    
    for (int x : arr)
        cout << x << " ";
    
    return 0;
}
```

Time Complexity: O(n log n) - Always, stable sort
Space Complexity: O(n) - Temporary arrays for merging
