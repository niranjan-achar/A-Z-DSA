Problem 13: Maximum Subarray Sum (Kadane using D&C)

Description:
Find the maximum sum of a contiguous subarray using divide and conquer approach.

Approach:
Divide array into two halves. Recursively find max sum in left, right, and crossing mid. Return maximum.

Brute Force Note:
Check all subarrays: O(n³) time.

C++ Code:
```cpp
#include <iostream>
#include <vector>
#include <climits>
using namespace std;

long long maxCrossingSum(vector<int>& arr, int low, int mid, int high) {
    long long leftSum = LLONG_MIN, sum = 0;
    for (int i = mid; i >= low; i--) {
        sum += arr[i];
        leftSum = max(leftSum, sum);
    }
    
    long long rightSum = LLONG_MIN;
    sum = 0;
    for (int i = mid + 1; i <= high; i++) {
        sum += arr[i];
        rightSum = max(rightSum, sum);
    }
    
    return leftSum + rightSum;
}

long long maxSubarraySum(vector<int>& arr, int low, int high) {
    if (low == high)
        return arr[low];
    
    int mid = low + (high - low) / 2;
    
    long long leftMax = maxSubarraySum(arr, low, mid);
    long long rightMax = maxSubarraySum(arr, mid + 1, high);
    long long crossMax = maxCrossingSum(arr, low, mid, high);
    
    return max({leftMax, rightMax, crossMax});
}

int main() {
    vector<int> arr = {-2, 1, -3, 4, -1, 2, 1, -5, 4};
    cout << "Max subarray sum: " << maxSubarraySum(arr, 0, arr.size() - 1) << endl;
    
    return 0;
}
```

Time Complexity: O(n log n) - Divide and conquer
Space Complexity: O(log n) - Recursion stack
