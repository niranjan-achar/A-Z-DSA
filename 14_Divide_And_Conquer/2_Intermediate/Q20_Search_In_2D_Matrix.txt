Problem 20: Search in 2D Matrix

Description:
Search for a target value in a 2D matrix where each row and column is sorted.

Approach:
Start from top-right or bottom-left corner. Move left or down/up based on comparison with target.

Brute Force Note:
Linear search through all elements: O(m*n) time.

C++ Code:
```cpp
#include <iostream>
#include <vector>
using namespace std;

bool searchMatrix(vector<vector<int>>& matrix, int target) {
    if (matrix.empty() || matrix[0].empty())
        return false;
    
    int row = 0;
    int col = matrix[0].size() - 1;
    
    while (row < matrix.size() && col >= 0) {
        if (matrix[row][col] == target) {
            return true;
        } else if (matrix[row][col] > target) {
            col--;
        } else {
            row++;
        }
    }
    
    return false;
}

int main() {
    vector<vector<int>> matrix = {
        {1, 4, 7, 11, 15},
        {2, 5, 8, 12, 19},
        {3, 6, 9, 16, 22},
        {10, 13, 14, 17, 24},
        {18, 21, 23, 26, 30}
    };
    
    int target = 13;
    
    if (searchMatrix(matrix, target)) {
        cout << "Element found" << endl;
    } else {
        cout << "Element not found" << endl;
    }
    
    return 0;
}
```

Time Complexity: O(m + n) - Move at most m rows and n columns
Space Complexity: O(1) - Constant space
