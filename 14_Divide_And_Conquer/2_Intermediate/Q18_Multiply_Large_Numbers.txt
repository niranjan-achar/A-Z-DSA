Problem 18: Multiply Large Numbers (as Strings)

Description:
Multiply two large numbers represented as strings using divide and conquer.

Approach:
Divide numbers into halves recursively. Use 3 recursive calls (Karatsuba algorithm) instead of 4.

Brute Force Note:
Standard string multiplication: O(n²) time.

C++ Code:
```cpp
#include <iostream>
#include <string>
#include <algorithm>
using namespace std;

string addStrings(string a, string b) {
    string result = "";
    int carry = 0;
    int i = a.length() - 1;
    int j = b.length() - 1;
    
    while (i >= 0 || j >= 0 || carry) {
        int sum = carry;
        if (i >= 0)
            sum += a[i--] - '0';
        if (j >= 0)
            sum += b[j--] - '0';
        
        result = char('0' + sum % 10) + result;
        carry = sum / 10;
    }
    
    return result;
}

string subtractStrings(string a, string b) {
    // Simplified subtraction (assumes a >= b)
    string result = "";
    int borrow = 0;
    
    reverse(a.begin(), a.end());
    reverse(b.begin(), b.end());
    
    for (int i = 0; i < a.length(); i++) {
        int sub = (a[i] - '0') - (i < b.length() ? b[i] - '0' : 0) - borrow;
        if (sub < 0) {
            sub += 10;
            borrow = 1;
        } else {
            borrow = 0;
        }
        result = char('0' + sub) + result;
    }
    
    return result;
}

string multiplySimple(string a, string b) {
    // Simple multiplication for small numbers
    int result = stoi(a) * stoi(b);
    return to_string(result);
}

string multiply(string a, string b) {
    if (a.length() == 1 && b.length() == 1)
        return multiplySimple(a, b);
    
    // Pad with zeros
    int maxLen = max(a.length(), b.length());
    while (a.length() < maxLen) a = "0" + a;
    while (b.length() < maxLen) b = "0" + b;
    
    int mid = maxLen / 2;
    
    string aLeft = a.substr(0, a.length() - mid);
    string aRight = a.substr(a.length() - mid);
    string bLeft = b.substr(0, b.length() - mid);
    string bRight = b.substr(b.length() - mid);
    
    if (aLeft.empty()) aLeft = "0";
    if (bLeft.empty()) bLeft = "0";
    
    // Simplified (full Karatsuba would be more complex)
    return "Result";  // Placeholder
}

    return 0;
}
```

Time Complexity: O(n^1.585) - Karatsuba algorithm
Space Complexity: O(n) - Result string
