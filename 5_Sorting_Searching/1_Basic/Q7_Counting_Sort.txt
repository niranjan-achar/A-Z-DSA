Problem 7: Counting Sort

Description:
Sort an array of non-negative integers using the Counting Sort algorithm which counts frequency of each element.

Approach:
Create a count array to store frequency of each element. Iterate through count array and reconstruct sorted array.

Brute Force Note:
Comparison-based sorts: O(n log n) time.

C++ Code:
```cpp
#include <iostream>
#include <vector>
using namespace std;

void countingSort(vector<int>& arr) {
    if (arr.empty())
        return;
    
    int max = *max_element(arr.begin(), arr.end());
    int min = *min_element(arr.begin(), arr.end());
    
    int range = max - min + 1;
    vector<int> count(range, 0);
    
    // Store count of each element
    for (int num : arr)
        count[num - min]++;
    
    // Change count array so that count[i] contains actual position
    for (int i = 1; i < range; i++)
        count[i] += count[i - 1];
    
    vector<int> output(arr.size());
    
    // Build output array
    for (int i = arr.size() - 1; i >= 0; i--) {
        output[count[arr[i] - min] - 1] = arr[i];
        count[arr[i] - min]--;
    }
    
    // Copy sorted elements back to arr
    for (int i = 0; i < arr.size(); i++)
        arr[i] = output[i];
}

int main() {
    vector<int> arr = {4, 2, 2, 8, 3, 3, 1};
    
    cout << "Original array: ";
    for (int x : arr)
        cout << x << " ";
    cout << endl;
    
    countingSort(arr);
    
    cout << "Sorted array: ";
    for (int x : arr)
        cout << x << " ";
    cout << endl;
    
    return 0;
}
```

Time Complexity: O(n + k) where k is range of input
Space Complexity: O(n + k) - Count array
