Question Number: 4
Problem Title: Sort array in ascending/descending order
Problem Description:
Given an array, sort it in ascending order and also in descending order.

Approach / Explanation:
1) Use std::sort for ascending order.
2) For descending, use a custom comparator or reverse the ascending result.
3) This provides efficient O(n log n) sorting.
4) Works for any comparable type.
5) Outputs both orders.
6) Keeps code concise and reliable.
7) Use long long for safety.

Brute Force Note:
Using O(n^2) sorts is slower for large inputs.

C++ Code:
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n; if (!(cin >> n) || n <= 0) return 0;
    vector<long long> a(n), b;
    for (int i = 0; i < n; ++i) cin >> a[i];
    b = a;

    sort(a.begin(), a.end());
    sort(b.begin(), b.end(), greater<long long>());

    for (int i = 0; i < n; ++i) {
        if (i) cout << " ";
        cout << a[i];
    }
    cout << "\n";
    for (int i = 0; i < n; ++i) {
        if (i) cout << " ";
        cout << b[i];
    }
    return 0;
}

Time Complexity: O(n log n)
Space Complexity: O(1) extra
