Question Number: 18
Problem Title: Check if array can be sorted by reversing a subarray
Problem Description:
Determine if the array can be sorted by reversing exactly one subarray.

Approach / Explanation:
1) Find the first index where order breaks (a[i] > a[i+1]).
2) Find the last index of this decreasing segment.
3) Reverse that segment and check if array becomes sorted.
4) This tests one possible reversal.
5) If already sorted, answer is true.
6) Linear scan plus reverse.
7) Output 1 if possible, 0 otherwise.

Brute Force Note:
Trying all subarray reversals costs O(n^3).

C++ Code:
#include <bits/stdc++.h>
using namespace std;

bool isSorted(const vector<long long>& a) {
    for (int i = 0; i + 1 < (int)a.size(); ++i) if (a[i] > a[i + 1]) return false;
    return true;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n; if (!(cin >> n) || n <= 0) return 0;
    vector<long long> a(n), b;
    for (int i = 0; i < n; ++i) cin >> a[i];

    if (isSorted(a)) { cout << 1; return 0; }

    int l = 0;
    while (l + 1 < n && a[l] <= a[l + 1]) ++l;
    int r = l + 1;
    while (r + 1 < n && a[r] > a[r + 1]) ++r;

    b = a;
    reverse(b.begin() + l, b.begin() + r + 1);

    cout << (isSorted(b) ? 1 : 0);
    return 0;
}

Time Complexity: O(n)
Space Complexity: O(n)
