Question Number: 27
Problem Title: Graph coloring with k colors
Problem Description:
Given a graph and k, determine if the graph can be colored with k colors.

Approach / Explanation:
1) Use backtracking to assign colors to vertices.
2) For each vertex, try colors 1..k.
3) Check if any neighbor has the same color.
4) If valid, recurse to next vertex.
5) If all vertices colored, success.
6) Backtrack on conflicts.
7) Output 1 if possible, else 0.

Brute Force Note:
Trying all colorings is k^n.

C++ Code:
#include <bits/stdc++.h>
using namespace std;

bool safe(int v, int c, const vector<vector<int>>& g, const vector<int>& color) {
    for (int u : g[v]) if (color[u] == c) return false;
    return true;
}

bool dfs(int v, int k, const vector<vector<int>>& g, vector<int>& color) {
    if (v == (int)g.size()) return true;
    for (int c = 1; c <= k; ++c) {
        if (!safe(v, c, g, color)) continue;
        color[v] = c;
        if (dfs(v + 1, k, g, color)) return true;
        color[v] = 0;
    }
    return false;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, m, k; if (!(cin >> n >> m >> k) || n <= 0) return 0;
    vector<vector<int>> g(n);
    for (int i = 0; i < m; ++i) {
        int u, v; cin >> u >> v; --u; --v;
        g[u].push_back(v);
        g[v].push_back(u);
    }
    vector<int> color(n, 0);
    cout << (dfs(0, k, g, color) ? 1 : 0);
    return 0;
}

Time Complexity: O(k^n)
Space Complexity: O(n+m)
