Question Number: 32
Problem Title: Find common elements in three sorted arrays
Problem Description:
Given three sorted arrays, find all elements common to all three.

Approach / Explanation:
1) Use three pointers i, j, k starting at 0.
2) If a[i] == b[j] == c[k], record it and advance all.
3) Otherwise, advance the pointer with the smallest value.
4) This preserves linear time in total length.
5) Skip duplicates if you want unique results.
6) Works because arrays are sorted.
7) Output all common elements.

Brute Force Note:
Using sets or nested loops costs more time or space.

C++ Code:
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n1, n2, n3; if (!(cin >> n1 >> n2 >> n3)) return 0;
    vector<long long> a(n1), b(n2), c(n3);
    for (int i = 0; i < n1; ++i) cin >> a[i];
    for (int i = 0; i < n2; ++i) cin >> b[i];
    for (int i = 0; i < n3; ++i) cin >> c[i];

    int i = 0, j = 0, k = 0;
    vector<long long> res;
    while (i < n1 && j < n2 && k < n3) {
        if (a[i] == b[j] && b[j] == c[k]) {
            if (res.empty() || res.back() != a[i]) res.push_back(a[i]);
            ++i; ++j; ++k;
        } else {
            long long mn = min({a[i], b[j], c[k]});
            if (a[i] == mn) ++i;
            else if (b[j] == mn) ++j;
            else ++k;
        }
    }

    if (res.empty()) {
        cout << -1;
    } else {
        for (int idx = 0; idx < (int)res.size(); ++idx) {
            if (idx) cout << " ";
            cout << res[idx];
        }
    }
    return 0;
}

Time Complexity: O(n1 + n2 + n3)
Space Complexity: O(1) extra
