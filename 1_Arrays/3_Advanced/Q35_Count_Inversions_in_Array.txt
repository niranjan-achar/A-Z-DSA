Question Number: 35
Problem Title: Count inversions in an array
Problem Description:
An inversion is a pair (i, j) with i < j and a[i] > a[j]. Count all inversions.

Approach / Explanation:
1) Use merge sort to count inversions during merging.
2) When a right element is placed before left, it forms inversions.
3) Add the number of remaining elements in left to the count.
4) Merge sort ensures O(n log n) time.
5) This avoids O(n^2) pair checks.
6) Use long long for the inversion count.
7) Output the total inversions.

Brute Force Note:
Checking all pairs costs O(n^2).

C++ Code:
#include <bits/stdc++.h>
using namespace std;

long long mergeCount(vector<long long>& a, vector<long long>& temp, int l, int m, int r) {
    long long inv = 0;
    int i = l, j = m + 1, k = l;
    while (i <= m && j <= r) {
        if (a[i] <= a[j]) temp[k++] = a[i++];
        else {
            temp[k++] = a[j++];
            inv += (m - i + 1);
        }
    }
    while (i <= m) temp[k++] = a[i++];
    while (j <= r) temp[k++] = a[j++];
    for (int t = l; t <= r; ++t) a[t] = temp[t];
    return inv;
}

long long sortCount(vector<long long>& a, vector<long long>& temp, int l, int r) {
    if (l >= r) return 0;
    int m = l + (r - l) / 2;
    long long inv = 0;
    inv += sortCount(a, temp, l, m);
    inv += sortCount(a, temp, m + 1, r);
    inv += mergeCount(a, temp, l, m, r);
    return inv;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n; if (!(cin >> n) || n <= 0) return 0;
    vector<long long> a(n), temp(n);
    for (int i = 0; i < n; ++i) cin >> a[i];

    cout << sortCount(a, temp, 0, n - 1);
    return 0;
}

Time Complexity: O(n log n)
Space Complexity: O(n)
