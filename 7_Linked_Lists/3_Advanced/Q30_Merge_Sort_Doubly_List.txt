Question Number: 30
Problem Title: Merge sort a doubly linked list
Problem Description:
Given a doubly linked list, sort it using merge sort.

Approach / Explanation:
1) Use slow/fast pointers to split the list.
2) Recursively sort each half.
3) Merge two sorted doubly lists, fixing prev pointers.
4) Merge sort is stable and efficient.
5) O(n log n) time with O(log n) recursion stack.
6) Works for doubly linked lists.
7) Output the sorted list.

Brute Force Note:
Copying values to array and sorting uses extra space.

C++ Code:
#include <bits/stdc++.h>
using namespace std;

struct Node { long long val; Node* prev; Node* next; Node(long long v): val(v), prev(nullptr), next(nullptr) {} };

Node* splitMid(Node* head) {
    Node* slow = head; Node* fast = head;
    while (fast && fast->next && fast->next->next) {
        slow = slow->next; fast = fast->next->next;
    }
    Node* second = slow->next;
    if (second) second->prev = nullptr;
    slow->next = nullptr;
    return second;
}

Node* mergeLists(Node* a, Node* b) {
    if (!a) return b; if (!b) return a;
    if (a->val <= b->val) {
        a->next = mergeLists(a->next, b);
        if (a->next) a->next->prev = a;
        a->prev = nullptr;
        return a;
    } else {
        b->next = mergeLists(a, b->next);
        if (b->next) b->next->prev = b;
        b->prev = nullptr;
        return b;
    }
}

Node* mergeSort(Node* head) {
    if (!head || !head->next) return head;
    Node* second = splitMid(head);
    head = mergeSort(head);
    second = mergeSort(second);
    return mergeLists(head, second);
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n; if (!(cin >> n) || n <= 0) return 0;
    Node* head = nullptr; Node* tail = nullptr;
    for (int i = 0; i < n; ++i) {
        long long x; cin >> x;
        Node* node = new Node(x);
        if (!head) head = tail = node; else { tail->next = node; node->prev = tail; tail = node; }
    }

    head = mergeSort(head);
    Node* cur = head; bool first = true;
    while (cur) { if (!first) cout << " "; cout << cur->val; first = false; cur = cur->next; }
    return 0;
}

Time Complexity: O(n log n)
Space Complexity: O(log n) recursion
