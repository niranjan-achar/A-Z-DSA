Question Number: 24
Problem Title: Segregate odd and even valued nodes in a linked list
Problem Description:
Given a linked list, rearrange nodes so that all odd values come before even values.

Approach / Explanation:
1) Use two dummy heads: one for odds and one for evens.
2) Traverse and append each node to the respective list.
3) Connect the odd list to the even list.
4) Ensure the last even node points to null.
5) This keeps relative order within odd and even nodes.
6) O(n) time and O(1) extra space.
7) Output the reordered list.

Brute Force Note:
Creating a new list with extra nodes uses more space.

C++ Code:
#include <bits/stdc++.h>
using namespace std;

struct Node { long long val; Node* next; Node(long long v): val(v), next(nullptr) {} };

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n; if (!(cin >> n) || n <= 0) return 0;
    Node* head = nullptr; Node* tail = nullptr;
    for (int i = 0; i < n; ++i) {
        long long x; cin >> x;
        Node* node = new Node(x);
        if (!head) head = tail = node; else { tail->next = node; tail = node; }
    }

    Node oddDummy(0), evenDummy(0);
    Node* oddTail = &oddDummy; Node* evenTail = &evenDummy;
    Node* cur = head;
    while (cur) {
        if (cur->val % 2 != 0) { oddTail->next = cur; oddTail = cur; }
        else { evenTail->next = cur; evenTail = cur; }
        cur = cur->next;
    }
    oddTail->next = evenDummy.next;
    evenTail->next = nullptr;

    cur = oddDummy.next; bool first = true;
    while (cur) { if (!first) cout << " "; cout << cur->val; first = false; cur = cur->next; }
    return 0;
}

Time Complexity: O(n)
Space Complexity: O(1)
