Question Number: 18
Problem Title: Clone a linked list with random pointers
Problem Description:
Given a linked list with next and random pointers, create a deep copy.

Approach / Explanation:
1) Interleave copied nodes with original nodes.
2) Set random pointers for copies using original->random->next.
3) Separate the two lists by fixing next pointers.
4) This avoids using extra hash maps.
5) O(n) time and O(1) extra space.
6) Works for any random structure.
7) Output the head of the cloned list.

Brute Force Note:
Using a hash map is simpler but uses O(n) space.

C++ Code:
#include <bits/stdc++.h>
using namespace std;

struct Node {
    long long val; Node* next; Node* random;
    Node(long long v): val(v), next(nullptr), random(nullptr) {}
};

Node* copyRandomList(Node* head) {
    if (!head) return nullptr;
    Node* cur = head;
    while (cur) {
        Node* copy = new Node(cur->val);
        copy->next = cur->next;
        cur->next = copy;
        cur = copy->next;
    }
    cur = head;
    while (cur) {
        if (cur->random) cur->next->random = cur->random->next;
        cur = cur->next->next;
    }
    Node dummy(0); Node* tail = &dummy;
    cur = head;
    while (cur) {
        Node* copy = cur->next;
        cur->next = copy->next;
        tail->next = copy; tail = copy;
        cur = cur->next;
    }
    return dummy.next;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    Node* head = nullptr;
    Node* clone = copyRandomList(head);
    (void)clone;
    return 0;
}

Time Complexity: O(n)
Space Complexity: O(1) extra
