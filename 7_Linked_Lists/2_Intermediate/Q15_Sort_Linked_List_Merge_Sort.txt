Question Number: 15
Problem Title: Sort a linked list (Merge Sort)
Problem Description:
Given a linked list, sort it in ascending order.

Approach / Explanation:
1) Use merge sort to sort the list.
2) Split the list into two halves using slow/fast pointers.
3) Recursively sort each half.
4) Merge two sorted lists.
5) Merge sort is stable for linked lists.
6) O(n log n) time.
7) Output the sorted list.

Brute Force Note:
Copying to array and sorting uses extra space.

C++ Code:
#include <bits/stdc++.h>
using namespace std;

struct Node { long long val; Node* next; Node(long long v): val(v), next(nullptr) {} };

Node* mergeLists(Node* a, Node* b) {
    Node dummy(0); Node* tail = &dummy;
    while (a && b) {
        if (a->val <= b->val) { tail->next = a; a = a->next; }
        else { tail->next = b; b = b->next; }
        tail = tail->next;
    }
    tail->next = a ? a : b;
    return dummy.next;
}

Node* splitMid(Node* head) {
    Node* slow = head; Node* fast = head; Node* prev = nullptr;
    while (fast && fast->next) { prev = slow; slow = slow->next; fast = fast->next->next; }
    if (prev) prev->next = nullptr;
    return slow;
}

Node* mergeSort(Node* head) {
    if (!head || !head->next) return head;
    Node* mid = splitMid(head);
    Node* left = mergeSort(head);
    Node* right = mergeSort(mid);
    return mergeLists(left, right);
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n; if (!(cin >> n) || n <= 0) return 0;
    Node* head = nullptr; Node* tail = nullptr;
    for (int i = 0; i < n; ++i) {
        long long x; cin >> x;
        Node* node = new Node(x);
        if (!head) head = tail = node; else { tail->next = node; tail = node; }
    }

    head = mergeSort(head);

    Node* cur = head; bool first = true;
    while (cur) { if (!first) cout << " "; cout << cur->val; first = false; cur = cur->next; }
    return 0;
}

Time Complexity: O(n log n)
Space Complexity: O(log n) recursion
