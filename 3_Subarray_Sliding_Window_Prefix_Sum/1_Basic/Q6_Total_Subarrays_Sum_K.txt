Question Number: 6
Problem Title: Find total number of subarrays with sum = K
Problem Description:
Given an array and K, count subarrays whose sum equals K.

Approach / Explanation:
1) Use prefix sums and a frequency map.
2) For each prefix sum, add freq[prefix - K] to the answer.
3) Then increment freq[prefix].
4) Initialize freq[0] = 1 for subarrays starting at index 0.
5) Works with negative values.
6) Single pass gives linear time.
7) Output the total count.

Brute Force Note:
Checking all subarrays costs O(n^2).

C++ Code:
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n; long long K;
    if (!(cin >> n) || n <= 0) return 0;
    vector<long long> a(n);
    for (int i = 0; i < n; ++i) cin >> a[i];
    cin >> K;

    unordered_map<long long, long long> freq;
    freq[0] = 1;
    long long prefix = 0, ans = 0;
    for (long long x : a) {
        prefix += x;
        long long need = prefix - K;
        if (freq.count(need)) ans += freq[need];
        ++freq[prefix];
    }

    cout << ans;
    return 0;
}

Time Complexity: O(n) average
Space Complexity: O(n)
