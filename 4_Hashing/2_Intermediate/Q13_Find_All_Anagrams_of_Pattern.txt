Question Number: 13
Problem Title: Find all anagrams of a pattern in a string
Problem Description:
Given a text and pattern, output starting indices of all anagram substrings.

Approach / Explanation:
1) Use sliding window of length m.
2) Maintain frequency counts for pattern and window.
3) Track matches with a counter for efficient comparison.
4) When counts match, record the start index.
5) Slide by removing left char and adding right char.
6) Linear time with fixed alphabet.
7) Output indices.

Brute Force Note:
Sorting each window costs O(n * m log m).

C++ Code:
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    string s, p;
    if (!getline(cin >> ws, s)) return 0;
    if (!getline(cin, p)) return 0;

    int n = (int)s.size(), m = (int)p.size();
    if (m == 0 || m > n) return 0;

    vector<int> need(256, 0), win(256, 0);
    for (unsigned char c : p) ++need[c];

    int matches = 0;
    for (int i = 0; i < 256; ++i) if (need[i] == 0) ++matches;

    vector<int> res;
    for (int i = 0; i < n; ++i) {
        unsigned char c = s[i];
        ++win[c];
        if (win[c] == need[c]) ++matches;
        else if (win[c] == need[c] + 1) --matches;

        if (i >= m) {
            unsigned char d = s[i - m];
            --win[d];
            if (win[d] == need[d]) ++matches;
            else if (win[d] == need[d] - 1) --matches;
        }

        if (i >= m - 1 && matches == 256) res.push_back(i - m + 1);
    }

    for (int i = 0; i < (int)res.size(); ++i) {
        if (i) cout << " ";
        cout << res[i];
    }
    return 0;
}

Time Complexity: O(n)
Space Complexity: O(1)
