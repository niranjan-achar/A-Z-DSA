Question Number: 14
Problem Title: KMP algorithm for pattern matching
Problem Description:
Given text and pattern, find all occurrences using KMP.

Approach / Explanation:
1) Build LPS array for the pattern.
2) Use two pointers to scan text and pattern.
3) On match, advance both; on mismatch, use LPS to shift pattern.
4) When pattern completes, record index.
5) This avoids re-checking characters.
6) Linear time algorithm.
7) Output indices separated by spaces.

Brute Force Note:
Naive matching is O(n*m).

C++ Code:
#include <bits/stdc++.h>
using namespace std;

vector<int> buildLPS(const string& p) {
    int m = p.size();
    vector<int> lps(m, 0);
    for (int i = 1, len = 0; i < m; ) {
        if (p[i] == p[len]) lps[i++] = ++len;
        else if (len) len = lps[len - 1];
        else lps[i++] = 0;
    }
    return lps;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    string text, pat; if (!(cin >> text >> pat)) return 0;
    int n = text.size(), m = pat.size();
    if (m == 0) { cout << 0; return 0; }
    vector<int> lps = buildLPS(pat);

    vector<int> ans;
    for (int i = 0, j = 0; i < n; ) {
        if (text[i] == pat[j]) { ++i; ++j; }
        if (j == m) { ans.push_back(i - j); j = lps[j - 1]; }
        else if (i < n && text[i] != pat[j]) {
            if (j) j = lps[j - 1];
            else ++i;
        }
    }

    if (ans.empty()) { cout << -1; return 0; }
    for (int i = 0; i < (int)ans.size(); ++i) { if (i) cout << " "; cout << ans[i]; }
    return 0;
}

Time Complexity: O(n + m)
Space Complexity: O(m)
