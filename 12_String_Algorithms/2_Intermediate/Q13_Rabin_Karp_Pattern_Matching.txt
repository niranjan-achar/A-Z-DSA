Question Number: 13
Problem Title: Rabin-Karp algorithm for pattern matching
Problem Description:
Given text and pattern, find all starting indices where pattern occurs.

Approach / Explanation:
1) Use rolling hash for the pattern and each window.
2) Precompute base powers.
3) Slide the window and update hash in O(1).
4) If hash matches, verify to avoid collisions.
5) Collect all matching indices.
6) Average linear time.
7) Output indices separated by spaces.

Brute Force Note:
Checking each window character-by-character is O(n*m).

C++ Code:
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    string text, pat; if (!(cin >> text >> pat)) return 0;
    int n = text.size(), m = pat.size();
    if (m == 0 || m > n) { cout << -1; return 0; }
    const long long MOD = 1000000007LL;
    const long long BASE = 911382323LL;

    long long hashP = 0, hashT = 0, powm = 1;
    for (int i = 0; i < m; ++i) {
        hashP = (hashP * BASE + (unsigned char)pat[i] + 1) % MOD;
        hashT = (hashT * BASE + (unsigned char)text[i] + 1) % MOD;
        if (i) powm = (powm * BASE) % MOD;
    }

    vector<int> ans;
    for (int i = 0; i + m <= n; ++i) {
        if (hashP == hashT) {
            bool ok = true;
            for (int j = 0; j < m; ++j) if (text[i+j] != pat[j]) { ok = false; break; }
            if (ok) ans.push_back(i);
        }
        if (i + m < n) {
            long long left = (unsigned char)text[i] + 1;
            long long right = (unsigned char)text[i+m] + 1;
            hashT = (hashT - left * powm) % MOD;
            if (hashT < 0) hashT += MOD;
            hashT = (hashT * BASE + right) % MOD;
        }
    }

    if (ans.empty()) { cout << -1; return 0; }
    for (int i = 0; i < (int)ans.size(); ++i) { if (i) cout << " "; cout << ans[i]; }
    return 0;
}

Time Complexity: O(n + m)
Space Complexity: O(1)
