Question Number: 16
Problem Title: Prim's Algorithm (Minimum Spanning Tree)
Problem Description:
Given a weighted undirected graph, compute the MST total weight.

Approach / Explanation:
1) Start from any node and push its edges into a min-heap.
2) Repeatedly pick the minimum weight edge to a new node.
3) Add that node and its edges to the heap.
4) Skip edges leading to already visited nodes.
5) Continue until all nodes are included.
6) Greedy choice ensures MST.
7) Output total weight.

Brute Force Note:
Checking all spanning trees is infeasible.

C++ Code:
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, m; if (!(cin >> n >> m) || n <= 0) return 0;
    vector<vector<pair<int,int>>> g(n);
    for (int i = 0; i < m; ++i) {
        int u, v, w; cin >> u >> v >> w; --u; --v;
        g[u].push_back({v, w});
        g[v].push_back({u, w});
    }

    vector<int> vis(n, 0);
    priority_queue<pair<int,int>, vector<pair<int,int>>, greater<pair<int,int>>> pq;
    pq.push({0, 0});
    long long total = 0; int used = 0;
    while (!pq.empty()) {
        auto [w, u] = pq.top(); pq.pop();
        if (vis[u]) continue;
        vis[u] = 1; total += w; ++used;
        for (auto [v, wt] : g[u]) if (!vis[v]) pq.push({wt, v});
    }

    cout << (used == n ? total : -1);
    return 0;
}

Time Complexity: O(m log n)
Space Complexity: O(n + m)
