Question Number: 30
Problem Title: Eulerian Path and Circuit
Problem Description:
Given an undirected graph, determine if it has an Eulerian path or circuit.

Approach / Explanation:
1) For Eulerian circuit: all vertices with non-zero degree are connected and all degrees are even.
2) For Eulerian path: exactly two vertices have odd degree and graph is connected.
3) Use DFS from a vertex with non-zero degree to check connectivity.
4) Count odd degree vertices.
5) Output 2 for circuit, 1 for path, 0 for none.
6) Works for undirected graphs.
7) Output the status.

Brute Force Note:
No brute force needed.

C++ Code:
#include <bits/stdc++.h>
using namespace std;

void dfs(int u, const vector<vector<int>>& g, vector<int>& vis) {
    vis[u] = 1;
    for (int v : g[u]) if (!vis[v]) dfs(v, g, vis);
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, m; if (!(cin >> n >> m) || n <= 0) return 0;
    vector<vector<int>> g(n);
    vector<int> deg(n, 0);
    for (int i = 0; i < m; ++i) {
        int u, v; cin >> u >> v; --u; --v;
        g[u].push_back(v); g[v].push_back(u);
        deg[u]++; deg[v]++;
    }

    int start = -1;
    for (int i = 0; i < n; ++i) if (deg[i] > 0) { start = i; break; }
    if (start == -1) { cout << 2; return 0; }

    vector<int> vis(n, 0);
    dfs(start, g, vis);
    for (int i = 0; i < n; ++i) if (deg[i] > 0 && !vis[i]) { cout << 0; return 0; }

    int odd = 0;
    for (int i = 0; i < n; ++i) if (deg[i] % 2 == 1) ++odd;
    if (odd == 0) cout << 2;
    else if (odd == 2) cout << 1;
    else cout << 0;
    return 0;
}

Time Complexity: O(n + m)
Space Complexity: O(n + m)
