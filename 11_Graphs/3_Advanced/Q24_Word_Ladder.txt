Question Number: 24
Problem Title: Word Ladder Problem
Problem Description:
Given begin and end words and a dictionary, find the shortest transformation length.

Approach / Explanation:
1) Use BFS from the begin word.
2) At each step, change one letter to generate neighbors.
3) If neighbor is in dictionary and unvisited, enqueue it.
4) First time reaching end gives shortest length.
5) Use a set for O(1) lookups.
6) BFS ensures shortest path.
7) Output the length or 0 if impossible.

Brute Force Note:
Trying all paths is exponential.

C++ Code:
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    string beginWord, endWord; if (!(cin >> beginWord >> endWord)) return 0;
    int n; cin >> n;
    unordered_set<string> dict;
    for (int i = 0; i < n; ++i) { string w; cin >> w; dict.insert(w); }
    if (!dict.count(endWord)) { cout << 0; return 0; }

    queue<pair<string,int>> q;
    q.push({beginWord, 1});
    unordered_set<string> vis;
    vis.insert(beginWord);

    while (!q.empty()) {
        auto [cur, d] = q.front(); q.pop();
        if (cur == endWord) { cout << d; return 0; }
        string temp = cur;
        for (int i = 0; i < (int)temp.size(); ++i) {
            char orig = temp[i];
            for (char ch = 'a'; ch <= 'z'; ++ch) {
                if (ch == orig) continue;
                temp[i] = ch;
                if (dict.count(temp) && !vis.count(temp)) {
                    vis.insert(temp);
                    q.push({temp, d + 1});
                }
            }
            temp[i] = orig;
        }
    }
    cout << 0;
    return 0;
}

Time Complexity: O(N * L * 26)
Space Complexity: O(N)
