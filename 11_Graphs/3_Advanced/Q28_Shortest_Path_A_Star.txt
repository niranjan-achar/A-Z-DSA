Question Number: 28
Problem Title: Shortest path using A* algorithm
Problem Description:
Given a grid with 0 as free and 1 as blocked, find shortest path length from start to goal.

Approach / Explanation:
1) Use A* with f = g + h (Manhattan heuristic).
2) g is cost from start, h is estimated to goal.
3) Use a priority queue ordered by f.
4) Relax neighbors and update g if a shorter path is found.
5) A* is optimal with admissible heuristic.
6) Works on grids with 4-directional moves.
7) Output shortest length or -1.

Brute Force Note:
Dijkstra explores more nodes without heuristic.

C++ Code:
#include <bits/stdc++.h>
using namespace std;

struct Node {
    int f, g, r, c;
    bool operator>(const Node& other) const { return f > other.f; }
};

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, m; if (!(cin >> n >> m) || n <= 0 || m <= 0) return 0;
    int sr, sc, tr, tc; cin >> sr >> sc >> tr >> tc;
    vector<vector<int>> grid(n, vector<int>(m));
    for (int i = 0; i < n; ++i) for (int j = 0; j < m; ++j) cin >> grid[i][j];

    auto h = [&](int r, int c){ return abs(r - tr) + abs(c - tc); };
    const int INF = 1e9;
    vector<vector<int>> dist(n, vector<int>(m, INF));
    priority_queue<Node, vector<Node>, greater<Node>> pq;
    dist[sr][sc] = 0;
    pq.push({h(sr, sc), 0, sr, sc});

    int dr[4] = {1, -1, 0, 0};
    int dc[4] = {0, 0, 1, -1};

    while (!pq.empty()) {
        Node cur = pq.top(); pq.pop();
        if (cur.r == tr && cur.c == tc) { cout << cur.g; return 0; }
        if (cur.g != dist[cur.r][cur.c]) continue;
        for (int k = 0; k < 4; ++k) {
            int nr = cur.r + dr[k], nc = cur.c + dc[k];
            if (nr < 0 || nc < 0 || nr >= n || nc >= m) continue;
            if (grid[nr][nc] == 1) continue;
            int ng = cur.g + 1;
            if (ng < dist[nr][nc]) {
                dist[nr][nc] = ng;
                pq.push({ng + h(nr, nc), ng, nr, nc});
            }
        }
    }
    cout << -1;
    return 0;
}

Time Complexity: O(n*m log(n*m))
Space Complexity: O(n*m)
