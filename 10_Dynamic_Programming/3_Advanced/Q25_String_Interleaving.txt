Question Number: 25
Problem Title: String Interleaving
Problem Description:
Given s1, s2, and s3, determine if s3 is an interleaving of s1 and s2.

Approach / Explanation:
1) If lengths do not add up, return false.
2) Use dp[i][j] for whether s3[0..i+j-1] can be formed.
3) Transition from dp[i-1][j] if s1[i-1] matches.
4) Transition from dp[i][j-1] if s2[j-1] matches.
5) Fill table row by row.
6) This reuses subproblems efficiently.
7) Output 1 if possible, else 0.

Brute Force Note:
Trying all interleavings is exponential.

C++ Code:
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    string s1, s2, s3; if (!(cin >> s1 >> s2 >> s3)) return 0;
    int n = (int)s1.size(), m = (int)s2.size();
    if ((int)s3.size() != n + m) { cout << 0; return 0; }
    vector<vector<char>> dp(n + 1, vector<char>(m + 1, 0));
    dp[0][0] = 1;
    for (int i = 0; i <= n; ++i) {
        for (int j = 0; j <= m; ++j) {
            if (i > 0 && s1[i-1] == s3[i+j-1]) dp[i][j] = dp[i][j] || dp[i-1][j];
            if (j > 0 && s2[j-1] == s3[i+j-1]) dp[i][j] = dp[i][j] || dp[i][j-1];
        }
    }
    cout << (dp[n][m] ? 1 : 0);
    return 0;
}

Time Complexity: O(n*m)
Space Complexity: O(n*m)
