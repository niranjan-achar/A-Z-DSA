Question Number: 21
Problem Title: Job Scheduling with Deadlines
Problem Description:
Given jobs with deadlines and profits, schedule to maximize total profit.

Approach / Explanation:
1) Sort jobs by profit descending.
2) Use DSU to find the latest available slot <= deadline.
3) If a slot is available, schedule the job and mark slot used.
4) Accumulate total profit and count of jobs.
5) Greedy profit-first ordering works with slot assignment.
6) DSU gives near O(1) slot lookup.
7) Output total jobs and total profit.

Brute Force Note:
Trying all job orders is factorial time.

C++ Code:
#include <bits/stdc++.h>
using namespace std;

struct Job { int d; long long p; };

struct DSU {
    vector<int> parent;
    DSU(int n): parent(n+1) { iota(parent.begin(), parent.end(), 0); }
    int find(int x){ return parent[x]==x? x : parent[x]=find(parent[x]); }
    void unite(int x,int y){ parent[find(x)] = find(y); }
};

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n; if (!(cin >> n) || n <= 0) return 0;
    vector<Job> jobs(n);
    int maxD = 0;
    for (int i = 0; i < n; ++i) { cin >> jobs[i].d >> jobs[i].p; maxD = max(maxD, jobs[i].d); }
    sort(jobs.begin(), jobs.end(), [](const Job& a, const Job& b){ return a.p > b.p; });

    DSU dsu(maxD);
    int count = 0; long long profit = 0;
    for (const auto& job : jobs) {
        int slot = dsu.find(job.d);
        if (slot > 0) {
            ++count; profit += job.p;
            dsu.unite(slot, slot - 1);
        }
    }
    cout << count << " " << profit;
    return 0;
}

Time Complexity: O(n log n)
Space Complexity: O(n)
