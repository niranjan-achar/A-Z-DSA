Question Number: 19
Problem Title: Modular Inverse
Problem Description:
Given a and mod, find the modular inverse of a under mod, if it exists.

Approach / Explanation:
1) Use extended Euclidean algorithm.
2) Inverse exists if gcd(a, mod) == 1.
3) Compute x such that ax + mod*y = 1.
4) x mod mod is the inverse.
5) Normalize to positive value.
6) O(log mod) time.
7) Output inverse or -1.

Brute Force Note:
Checking all x is O(mod).

C++ Code:
#include <bits/stdc++.h>
using namespace std;

long long egcd(long long a, long long b, long long &x, long long &y) {
    if (b == 0) { x = 1; y = 0; return a; }
    long long x1, y1;
    long long g = egcd(b, a % b, x1, y1);
    x = y1;
    y = x1 - (a / b) * y1;
    return g;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    long long a, mod; if (!(cin >> a >> mod) || mod == 0) return 0;
    long long x, y;
    long long g = egcd(a, mod, x, y);
    if (g != 1) { cout << -1; return 0; }
    long long inv = (x % mod + mod) % mod;
    cout << inv;
    return 0;
}

Time Complexity: O(log mod)
Space Complexity: O(1)
