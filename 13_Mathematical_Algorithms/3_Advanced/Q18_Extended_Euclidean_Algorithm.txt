Question Number: 18
Problem Title: Extended Euclidean Algorithm
Problem Description:
Given a and b, find x and y such that ax + by = gcd(a,b).

Approach / Explanation:
1) Use recursion to compute gcd and coefficients.
2) Base case: if b==0, x=1, y=0.
3) Recurse with (b, a%b).
4) Update x and y from recursion results.
5) This computes Bézout coefficients.
6) O(log min(a,b)) time.
7) Output gcd, x, y.

Brute Force Note:
No brute force needed.

C++ Code:
#include <bits/stdc++.h>
using namespace std;

long long egcd(long long a, long long b, long long &x, long long &y) {
    if (b == 0) { x = 1; y = 0; return a; }
    long long x1, y1;
    long long g = egcd(b, a % b, x1, y1);
    x = y1;
    y = x1 - (a / b) * y1;
    return g;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    long long a, b; if (!(cin >> a >> b)) return 0;
    long long x, y;
    long long g = egcd(a, b, x, y);
    cout << g << " " << x << " " << y;
    return 0;
}

Time Complexity: O(log min(a,b))
Space Complexity: O(1)
